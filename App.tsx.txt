import { useEffect, useMemo, useState } from "react";
import { categories, tools, type CategoryId, type Tool } from "./tools";
import { readArray, readString, writeValue } from "./storage";
import { Icon } from "./Icon";

type Theme = "light" | "dark";
type View = "home" | "all" | "favourites" | "recent";

function App() {
  const [theme, setTheme] = useState<Theme>(() => {
    const saved = readString("theme", "");
    if (saved === "light" || saved === "dark") return saved;
    return window.matchMedia?.("(prefers-color-scheme: dark)").matches ? "dark" : "light";
  });
  const [query, setQuery] = useState("");
  const [category, setCategory] = useState<CategoryId>("all");
  const [view, setView] = useState<View>("home");
  const [favourites, setFavourites] = useState<string[]>(() => readArray("favourites"));
  const [recent, setRecent] = useState<string[]>(() => readArray("recent"));
  const [mobileMenu, setMobileMenu] = useState(false);
  const [toast, setToast] = useState("");
  const [installPrompt, setInstallPrompt] = useState<BeforeInstallPromptEvent | null>(null);

  useEffect(() => {
    document.documentElement.dataset.theme = theme;
    document.documentElement.style.colorScheme = theme;
    writeValue("theme", theme);
  }, [theme]);

  useEffect(() => writeValue("favourites", favourites), [favourites]);
  useEffect(() => writeValue("recent", recent), [recent]);

  useEffect(() => {
    const handler = (event: Event) => {
      event.preventDefault();
      setInstallPrompt(event as BeforeInstallPromptEvent);
    };
    window.addEventListener("beforeinstallprompt", handler);
    return () => window.removeEventListener("beforeinstallprompt", handler);
  }, []);

  useEffect(() => {
    if (!toast) return;
    const timer = window.setTimeout(() => setToast(""), 3200);
    return () => window.clearTimeout(timer);
  }, [toast]);

  useEffect(() => {
    const handleShortcut = (event: KeyboardEvent) => {
      if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === "k") {
        event.preventDefault();
        document.querySelector<HTMLInputElement>(".search-box input")?.focus();
      }
    };
    window.addEventListener("keydown", handleShortcut);
    return () => window.removeEventListener("keydown", handleShortcut);
  }, []);

  const filteredTools = useMemo(() => {
    let list = [...tools];
    if (view === "favourites") list = list.filter((tool) => favourites.includes(tool.id));
    else if (view === "recent") list = list.filter((tool) => recent.includes(tool.id));
    if (category !== "all" && view !== "favourites" && view !== "recent") {
      list = list.filter((tool) => tool.category === category);
    }
    const normalized = query.trim().toLowerCase();
    if (normalized) {
      list = list.filter((tool) =>
        `${tool.name} ${tool.description} ${tool.category}`.toLowerCase().includes(normalized)
      );
    }
    if (view === "recent") list.sort((a, b) => recent.indexOf(a.id) - recent.indexOf(b.id));
    return list;
  }, [category, favourites, query, recent, view]);

  const selectedCategory = categories.find((item) => item.id === category);
  const title = view === "favourites" ? "Your favourites" : view === "recent" ? "Recently viewed" :
    view === "all" ? "All tools" : category === "all" ? "Explore tools" : selectedCategory?.label ?? "Explore tools";

  function selectView(next: View) {
    setView(next);
    setCategory("all");
    setQuery("");
    setMobileMenu(false);
  }

  function selectCategory(next: CategoryId) {
    setCategory(next);
    setView("all");
    setMobileMenu(false);
  }

  function toggleFavourite(toolId: string) {
    setFavourites((current) =>
      current.includes(toolId) ? current.filter((id) => id !== toolId) : [toolId, ...current]
    );
  }

  function previewTool(tool: Tool) {
    setRecent((current) => [tool.id, ...current.filter((id) => id !== tool.id)].slice(0, 12));
    setToast(`${tool.name} is planned for a future release. This preview does not pretend the tool is already working.`);
  }

  async function installApp() {
    if (!installPrompt) {
      setToast("To install FreeFile, use your browser menu and choose “Install app” or “Add to Home screen”.");
      return;
    }
    await installPrompt.prompt();
    setInstallPrompt(null);
  }

  function clearLocalData() {
    setFavourites([]);
    setRecent([]);
    try {
      localStorage.removeItem("freefile:v1:favourites");
      localStorage.removeItem("freefile:v1:recent");
    } catch {
      // Ignore unavailable storage.
    }
    setToast("Your FreeFile favourites and recent list have been cleared.");
  }

  const popularTools = tools.filter((tool) => tool.popular).slice(0, 4);

  return (
    <div className="app-shell">
      <header className="topbar">
        <a className="brand" href="#" onClick={(event) => { event.preventDefault(); selectView("home"); }} aria-label="FreeFile home">
          <span className="brand-mark"><Icon name="file-text" size={21} /></span>
          <span>FreeFile</span>
        </a>
        <nav className="desktop-nav" aria-label="Main navigation">
          <button className={view === "home" ? "nav-link active" : "nav-link"} onClick={() => selectView("home")}>Home</button>
          <button className={view === "all" ? "nav-link active" : "nav-link"} onClick={() => selectView("all")}>All tools</button>
          <button className={view === "favourites" ? "nav-link active" : "nav-link"} onClick={() => selectView("favourites")}>Favourites</button>
          <button className={view === "recent" ? "nav-link active" : "nav-link"} onClick={() => selectView("recent")}>Recently used</button>
        </nav>
        <div className="top-actions">
          <button className="icon-button install-button" onClick={installApp} aria-label="Install FreeFile" title="Install FreeFile">
            <Icon name="arrow" size={18}/><span>Install app</span>
          </button>
          <button className="icon-button" onClick={() => setTheme(theme === "light" ? "dark" : "light")} aria-label={`Switch to ${theme === "light" ? "dark" : "light"} mode`} title="Change theme">
            <Icon name={theme === "light" ? "moon" : "sun"} />
          </button>
          <button className="icon-button mobile-menu-button" onClick={() => setMobileMenu((open) => !open)} aria-label="Toggle menu">
            <Icon name={mobileMenu ? "close" : "menu"} />
          </button>
        </div>
      </header>

      {mobileMenu && (
        <nav className="mobile-nav" aria-label="Mobile navigation">
          <button onClick={() => selectView("home")}>Home</button>
          <button onClick={() => selectView("all")}>All tools</button>
          <button onClick={() => selectView("favourites")}>Favourites</button>
          <button onClick={() => selectView("recent")}>Recently used</button>
        </nav>
      )}

      <main>
        {view === "home" && category === "all" && (
          <>
            <section className="hero">
              <div className="hero-copy">
                <div className="eyebrow"><span className="eyebrow-dot" /> FREE TOOLS. NO ACCOUNT NEEDED.</div>
                <h1>Your files,<br/><span>made simple.</span></h1>
                <p>Convert, compress, edit and manage your files with a thoughtful collection of free online tools.</p>
                <div className="hero-actions">
                  <button className="button button-primary" onClick={() => selectView("all")}>Explore all tools <Icon name="arrow" size={17}/></button>
                  <span className="hero-note"><Icon name="shield" size={16}/> Private by design</span>
                </div>
              </div>
              <div className="hero-art" aria-hidden="true">
                <div className="art-glow"/>
                <div className="art-card art-card-back"><Icon name="image" size={27}/><span>image.png</span><small>2.4 MB</small></div>
                <div className="art-card art-card-front"><span className="pdf-tile"><Icon name="file-pdf" size={30}/></span><div><strong>document.pdf</strong><small>Ready when you are</small></div><span className="art-check">✓</span></div>
                <div className="art-mini"><Icon name="shield" size={17}/> Files stay yours</div>
              </div>
            </section>

            <section className="trust-strip" aria-label="FreeFile commitments">
              <div><Icon name="check-circle" size={17}/><span>Always free</span></div>
              <div><Icon name="shield" size={17}/><span>Privacy-first</span></div>
              <div><Icon name="droplet" size={17}/><span>No watermarks</span></div>
              <div><Icon name="user" size={17}/><span>No sign-up</span></div>
            </section>

            <section className="section-block">
              <div className="section-heading">
                <div><span className="section-kicker">FIND THE RIGHT TOOL</span><h2>What do you need to do?</h2></div>
                <button className="text-button" onClick={() => selectView("all")}>View all tools <Icon name="arrow" size={16}/></button>
              </div>
              <SearchBox query={query} setQuery={setQuery}/>
              <div className="category-grid">
                {categories.filter((item) => item.id !== "all").map((item) => (
                  <button className="category-card" key={item.id} onClick={() => selectCategory(item.id)}>
                    <span className={`category-icon ${item.id}`}><Icon name={item.icon}/></span>
                    <span className="category-copy"><strong>{item.label}</strong><small>{item.description}</small></span>
                    <Icon name="arrow" size={16} className="category-arrow"/>
                  </button>
                ))}
              </div>
            </section>

            <section className="section-block popular-section">
              <div className="section-heading">
                <div><span className="section-kicker">A GOOD PLACE TO START</span><h2>Popular tools</h2></div>
                <span className="muted-label">Planned for upcoming releases</span>
              </div>
              <ToolGrid items={popularTools} favourites={favourites} onFavourite={toggleFavourite} onPreview={previewTool}/>
            </section>
          </>
        )}

        {(view !== "home" || category !== "all" || query.trim()) && (
          <section className="directory-section">
            <div className="directory-top">
              <div>
                <span className="section-kicker">FREEFILE TOOLKIT</span>
                <h1 className="directory-title">{title}</h1>
                <p className="directory-description">{selectedCategory?.description ?? "Browse the tools planned for the FreeFile toolkit."}</p>
              </div>
              <span className="result-count">{filteredTools.length} {filteredTools.length === 1 ? "tool" : "tools"}</span>
            </div>
            <SearchBox query={query} setQuery={setQuery}/>
            <div className="filter-row" aria-label="Filter by category">
              {categories.map((item) => (
                <button key={item.id} className={category === item.id ? "filter-chip selected" : "filter-chip"} onClick={() => { setCategory(item.id); if (view === "home") setView("all"); }}>
                  <Icon name={item.icon} size={15}/>{item.label}
                </button>
              ))}
            </div>
            {filteredTools.length > 0 ? (
              <ToolGrid items={filteredTools} favourites={favourites} onFavourite={toggleFavourite} onPreview={previewTool}/>
            ) : (
              <div className="empty-state">
                <span className="empty-icon"><Icon name={view === "favourites" ? "heart" : "search"} size={25}/></span>
                <h3>{view === "favourites" ? "No favourites yet" : view === "recent" ? "Nothing viewed yet" : "No matching tools"}</h3>
                <p>{view === "favourites" ? "Tap the heart on a tool to save it here." : view === "recent" ? "Tools you preview will appear here." : "Try another search or choose a different category."}</p>
                <button className="button button-secondary" onClick={() => { setQuery(""); selectView("all"); }}>Browse all tools</button>
              </div>
            )}
          </section>
        )}

        <section className="privacy-panel">
          <div className="privacy-icon"><Icon name="shield" size={24}/></div>
          <div><span className="section-kicker">YOUR FILES, YOUR CONTROL</span><h2>Privacy isn't an extra feature.</h2><p>FreeFile is being built around browser-based processing. Tools will clearly explain when a file needs to leave your device. No account or cloud file locker is required.</p></div>
          <button className="privacy-link" onClick={() => setToast("Privacy details will be published with the first processing tools.")}>Our privacy promise <Icon name="arrow" size={16}/></button>
        </section>
      </main>

      <footer className="footer">
        <div className="footer-main">
          <a className="brand footer-brand" href="#" onClick={(event) => { event.preventDefault(); selectView("home"); }}><span className="brand-mark"><Icon name="file-text" size={19}/></span><span>FreeFile</span></a>
          <p>Simple tools for the files that matter.</p>
          <div className="footer-links"><button onClick={() => setToast("About FreeFile — a free, privacy-first file utility project.")}>About</button><button onClick={() => setToast("A full privacy policy will be added before processing tools launch.")}>Privacy</button><button onClick={() => setToast("Contact details will be added before public launch.")}>Contact</button></div>
        </div>
        <div className="footer-bottom"><span>© {new Date().getFullYear()} FreeFile</span><span>Free to use. Built with care.</span></div>
      </footer>

      <button className="clear-data-button" onClick={clearLocalData} title="Clear favourites and recently viewed tools">Clear local data</button>
      {toast && <div className="toast" role="status" aria-live="polite">{toast}<button onClick={() => setToast("")} aria-label="Dismiss message"><Icon name="close" size={15}/></button></div>}
    </div>
  );
}

function SearchBox({ query, setQuery }: { query: string; setQuery: (value: string) => void }) {
  return (
    <label className="search-box">
      <Icon name="search" size={19}/>
      <input value={query} onChange={(event) => setQuery(event.target.value)} placeholder="Search for a tool..." aria-label="Search tools"/>
      {query && <button type="button" onClick={() => setQuery("")} aria-label="Clear search"><Icon name="close" size={16}/></button>}
      <kbd>⌘ K</kbd>
    </label>
  );
}

function ToolGrid({ items, favourites, onFavourite, onPreview }: {
  items: Tool[];
  favourites: string[];
  onFavourite: (id: string) => void;
  onPreview: (tool: Tool) => void;
}) {
  return (
    <div className="tool-grid">
      {items.map((tool) => (
        <article className="tool-card" key={tool.id}>
          <button className="tool-main" onClick={() => onPreview(tool)} aria-label={`Preview ${tool.name}`}>
            <span className={`tool-icon ${tool.category}`}><Icon name={tool.icon}/></span>
            <span className="tool-info"><strong>{tool.name}</strong><small>{tool.description}</small></span>
            <span className="planned-tag">Planned</span>
          </button>
          <div className="tool-card-bottom">
            <span className="tool-category">{categories.find((item) => item.id === tool.category)?.label}</span>
            <button className={favourites.includes(tool.id) ? "favourite-button saved" : "favourite-button"} onClick={() => onFavourite(tool.id)} aria-label={favourites.includes(tool.id) ? `Remove ${tool.name} from favourites` : `Add ${tool.name} to favourites`} aria-pressed={favourites.includes(tool.id)}>
              <Icon name="heart" size={17}/>
            </button>
          </div>
        </article>
      ))}
    </div>
  );
}

type BeforeInstallPromptEvent = Event & {
  prompt: () => Promise<void>;
  userChoice?: Promise<{ outcome: "accepted" | "dismissed"; platform: string }>;
};

export default App;
